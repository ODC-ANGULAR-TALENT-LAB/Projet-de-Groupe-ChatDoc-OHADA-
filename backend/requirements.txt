# ChatDocs OHADA - dependances du backend
# Python 3.11 ou 3.12 (pas 3.14 : roues manquantes pour psycopg2).
#
# Installation :
#   py -3.11 -m venv .venv
#   .venv\Scripts\activate
#   pip install -r requirements.txt

# --- API -------------------------------------------------------------
pip install fastapi
pip install uvicorn[standard]
pip install python-multipart

# --- Base de donnees -------------------------------------------------
pip install sqlalchemy
pip install psycopg2-binary
pip install pgvector

# --- Configuration ---------------------------------------------------
pip install pydantic-settings
pip install email-validator     # requis par pydantic.EmailStr

# --- Generation de documents (phase F) -------------------------------
# Export Word des documents types. Le format .docx est celui que le
# client attend pour RETOUCHER le document ; le PDF, celui qu'il
# transmet. Les deux sont demandes par le cahier des charges (§5).
pip install python-docx

# --- Fournisseur LLM (phase E) ---------------------------------------
pip install anthropic

# --- Securite (phase G) ----------------------------------------------
pip install python-jose[cryptography]
pip install bcrypt              # directement, sans passlib : voir app/services/securite.py
pip install google-auth         # verification des jetons d'identite Google
pip install requests            # exige par google.auth.transport.requests

# --- Ingestion du corpus (phase B) -----------------------------------
pip install pdfplumber
pip install pytesseract
# Export PDF d'une reponse sourcee. Il etait deja present comme
# dependance transitive : on le declare, parce qu'on s'en sert
# directement et qu'une transitive peut disparaitre a la prochaine mise
# a jour de ce qui l'apportait.
pip install reportlab
# Le rendu des pages en images passe par pypdfium2, deja tire par
# pdfplumber : pas de poppler a installer ni a mettre dans le PATH.
# pdf2image n'est donc plus necessaire.
pip install pillow

# --- Appels sortants et tests ----------------------------------------
pip install httpx
pip install pytest
